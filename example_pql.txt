# Link Tracker PromQL examples

## RED metrics

# Request rate by application and URI.
sum by (application, uri) (
  rate(http_server_requests_seconds_count{application=~"$application"}[1m])
)

# 5xx error rate by application and URI.
sum by (application, uri) (
  rate(http_server_requests_seconds_count{application=~"$application", status=~"5.."}[1m])
)

# Error ratio by application.
sum by (application) (
  rate(http_server_requests_seconds_count{application=~"$application", status=~"5.."}[1m])
)
/
sum by (application) (
  rate(http_server_requests_seconds_count{application=~"$application"}[1m])
)

# HTTP request latency P50.
histogram_quantile(
  0.50,
  sum by (le, application, uri) (
    rate(http_server_requests_seconds_bucket{application=~"$application"}[5m])
  )
)

# HTTP request latency P95.
histogram_quantile(
  0.95,
  sum by (le, application, uri) (
    rate(http_server_requests_seconds_bucket{application=~"$application"}[5m])
  )
)

# HTTP request latency P99.
histogram_quantile(
  0.99,
  sum by (le, application, uri) (
    rate(http_server_requests_seconds_bucket{application=~"$application"}[5m])
  )
)

# JVM memory usage by application and memory area.
sum by (application, area) (
  jvm_memory_used_bytes{application=~"$application"}
)

## Business metrics

# User Telegram messages per second.
sum by (application, request_type) (
  rate(telegram_requests_total{application="bot"}[1m])
)
